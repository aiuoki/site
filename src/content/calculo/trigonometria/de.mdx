<Section eyebrow={"1 · Ausgangspunkt"} title={"Was der Einheitskreis ist"}>
  <Prose html={"Ein Kreis mit <b>Radius 1</b> um den Ursprung <span class=\"mr\">(0, 0)</span>. Ein Winkel <span class=\"m\">θ</span> wird von der positiven <span class=\"m\">x</span>-Achse aus gegen den Uhrzeigersinn gemessen. Wo der freie Schenkel des Winkels den Kreis schneidet, liegt der Punkt <span class=\"m\">P</span>, und seine Koordinaten sind genau <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sin <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Grad"} text={"Eine volle Umdrehung sind 360°."} />
    <DefItem title={"Bogenmaß (Radiant)"} text={"Der Winkel, dessen Bogen so lang ist wie der Radius. Eine Umdrehung sind <span class=\"mr\">2π</span> rad. Umrechnung: <span class=\"mr\">rad = Grad · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Quadrant"} text={"Jeder der 4 Bereiche, in die die Achsen die Ebene teilen, gegen den Uhrzeigersinn von I bis IV nummeriert."} />
    <DefItem title={"Bezugswinkel"} text={"Der spitze Winkel zwischen dem freien Schenkel und der <span class=\"m\">x</span>-Achse. Seine Funktionswerte sind die von <span class=\"m\">θ</span>, bis aufs Vorzeichen."} />
  </Defs>
</Section>

<Section eyebrow={"2 · Die sechs Funktionen"} title={"Was jede Strecke misst"} lead={"Da der Radius 1 ist, hat die Hypotenuse des Dreiecks die Länge 1, und jede Funktion wird zu einer sichtbaren Länge."}>
  <RatioGrid>
    <RatioCard ratio="sin" name={"Sinus"} tex={"\\Sin\\theta = y"} text={"<b>Senkrechte orange Strecke</b> von der <i>x</i>-Achse bis <i>P</i>. Sie ist die Höhe des Punkts. Immer zwischen −1 und 1."} />
    <RatioCard ratio="cos" name={"Kosinus"} tex={"\\Cos\\theta = x"} text={"<b>Waagrechte violette Strecke</b> vom Ursprung bis zum Schatten von <i>P</i> auf der <i>x</i>-Achse. Immer zwischen −1 und 1."} />
    <RatioCard ratio="tan" name={"Tangens"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Gelbe Strecke</b> auf der Geraden <i>x</i> = 1, die den Kreis berührt (daher der Name). Bei 90° und 270° nicht definiert, weil cos = 0."} />
    <RatioCard ratio="cot" name={"Kotangens"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Blaue Strecke</b> auf der Geraden <i>y</i> = 1. Bei 0° und 180° nicht definiert, weil sin = 0."} />
    <RatioCard ratio="sec" name={"Sekans"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Rote Strecke</b> vom Ursprung bis zum Ende des Tangens. Ihr Betrag ist nie kleiner als 1."} />
    <RatioCard ratio="csc" name={"Kosekans"} tex={"\\Csc\\theta = 1 \\mathbin{/} \\Sin\\theta"} text={"<b>Grüne Strecke</b> vom Ursprung bis zum Ende des Kotangens. Ihr Betrag ist nie kleiner als 1."} />
  </RatioGrid>
</Section>
