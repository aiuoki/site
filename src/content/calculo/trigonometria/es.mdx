<Section eyebrow={"1 · Punto de partida"} title={"Qué es el círculo unitario"}>
  <Prose html={"Es una circunferencia de <b>radio 1</b> centrada en el origen <span class=\"mr\">(0, 0)</span>. Un ángulo <span class=\"m\">θ</span> se mide desde el eje <span class=\"m\">x</span> positivo, girando en sentido antihorario. Donde el lado final del ángulo corta al círculo está el punto <span class=\"m\">P</span>, y sus coordenadas son justo <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sen <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Grado"} text={"Una vuelta completa son 360°."} />
    <DefItem title={"Radián"} text={"Ángulo cuyo arco mide lo mismo que el radio. Una vuelta son <span class=\"mr\">2π</span> rad. Para pasar: <span class=\"mr\">rad = grados · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Cuadrante"} text={"Cada una de las 4 zonas en que los ejes dividen el plano, numeradas I a IV en sentido antihorario."} />
    <DefItem title={"Ángulo de referencia"} text={"El ángulo agudo entre el lado final y el eje <span class=\"m\">x</span>. Sus razones son las mismas que las de <span class=\"m\">θ</span>, salvo el signo."} />
  </Defs>
</Section>

<Section eyebrow={"2 · Las seis razones"} title={"Qué mide cada segmento"} lead={"Al ser el radio 1, la hipotenusa del triángulo vale 1 y cada razón se convierte en una longitud que se puede ver."}>
  <RatioGrid>
    <RatioCard ratio="sin" name={"Seno"} tex={"\\Sin\\theta = y"} text={"<b>Segmento naranja vertical</b> desde el eje <i>x</i> hasta <i>P</i>. Es la altura del punto. Siempre entre −1 y 1."} />
    <RatioCard ratio="cos" name={"Coseno"} tex={"\\Cos\\theta = x"} text={"<b>Segmento morado horizontal</b> desde el origen hasta la sombra de <i>P</i> en el eje <i>x</i>. Siempre entre −1 y 1."} />
    <RatioCard ratio="tan" name={"Tangente"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento amarillo</b> sobre la recta <i>x</i> = 1, que es tangente al círculo (de ahí el nombre). No existe en 90° y 270°, porque cos = 0."} />
    <RatioCard ratio="cot" name={"Cotangente"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento azul</b> sobre la recta <i>y</i> = 1. No existe en 0° y 180°, porque sen = 0."} />
    <RatioCard ratio="sec" name={"Secante"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento rojo</b> desde el origen hasta el extremo de la tangente. Su valor absoluto nunca baja de 1."} />
    <RatioCard ratio="csc" name={"Cosecante"} tex={"\\Csc\\theta = 1 \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento verde</b> desde el origen hasta el extremo de la cotangente. Su valor absoluto nunca baja de 1."} />
  </RatioGrid>
</Section>
