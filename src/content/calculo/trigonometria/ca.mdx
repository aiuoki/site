<Section eyebrow={"1 · Punt de partida"} title={"Què és el cercle unitari"}>
  <Prose html={"És una circumferència de <b>radi 1</b> centrada a l'origen <span class=\"mr\">(0, 0)</span>. Un angle <span class=\"m\">θ</span> es mesura des de l'eix <span class=\"m\">x</span> positiu, girant en sentit antihorari. On el costat final de l'angle talla el cercle hi ha el punt <span class=\"m\">P</span>, i les seves coordenades són justament <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sin <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Grau"} text={"Una volta completa són 360°."} />
    <DefItem title={"Radiant"} text={"Angle l'arc del qual mesura el mateix que el radi. Una volta són <span class=\"mr\">2π</span> rad. Per passar-hi: <span class=\"mr\">rad = graus · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Quadrant"} text={"Cadascuna de les 4 zones en què els eixos divideixen el pla, numerades de I a IV en sentit antihorari."} />
    <DefItem title={"Angle de referència"} text={"L'angle agut entre el costat final i l'eix <span class=\"m\">x</span>. Les seves raons són les mateixes que les de <span class=\"m\">θ</span>, llevat del signe."} />
  </Defs>
</Section>

<Section eyebrow={"2 · Les sis raons"} title={"Què mesura cada segment"} lead={"Com que el radi és 1, la hipotenusa del triangle val 1 i cada raó es converteix en una longitud que es pot veure."}>

    <RatioCard ratio="sin" name={"Sinus"} tex={"\\Sin\\theta = y"} text={"<b>Segment taronja vertical</b> des de l'eix <i>x</i> fins a <i>P</i>. És l'altura del punt. Sempre entre −1 i 1."} />
    <RatioCard ratio="cos" name={"Cosinus"} tex={"\\Cos\\theta = x"} text={"<b>Segment morat horitzontal</b> des de l'origen fins a l'ombra de <i>P</i> sobre l'eix <i>x</i>. Sempre entre −1 i 1."} />
    <RatioCard ratio="tan" name={"Tangent"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Segment groc</b> sobre la recta <i>x</i> = 1, que és tangent al cercle (d'aquí el nom). No existeix a 90° i 270°, perquè cos = 0."} />
    <RatioCard ratio="cot" name={"Cotangent"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Segment blau</b> sobre la recta <i>y</i> = 1. No existeix a 0° i 180°, perquè sin = 0."} />
    <RatioCard ratio="sec" name={"Secant"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Segment vermell</b> des de l'origen fins a l'extrem de la tangent. El seu valor absolut mai baixa d'1."} />
