<Section eyebrow={"1 · Starting point"} title={"What the unit circle is"}>
  <Prose html={"A circle with <b>radius 1</b> centred at the origin <span class=\"mr\">(0, 0)</span>. An angle <span class=\"m\">θ</span> is measured from the positive <span class=\"m\">x</span>-axis, turning anticlockwise. Where the angle's terminal side crosses the circle lies the point <span class=\"m\">P</span>, and its coordinates are exactly <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sin <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Degree"} text={"A full turn is 360°."} />
    <DefItem title={"Radian"} text={"The angle whose arc is as long as the radius. A full turn is <span class=\"mr\">2π</span> rad. To convert: <span class=\"mr\">rad = degrees · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Quadrant"} text={"Each of the 4 regions the axes split the plane into, numbered I to IV anticlockwise."} />
    <DefItem title={"Reference angle"} text={"The acute angle between the terminal side and the <span class=\"m\">x</span>-axis. Its ratios equal those of <span class=\"m\">θ</span>, except for the sign."} />
  </Defs>
</Section>

<Section eyebrow={"2 · The six ratios"} title={"What each segment measures"} lead={"Since the radius is 1, the triangle's hypotenuse is 1 and every ratio becomes a length you can see."}>
  <RatioGrid>
    <RatioCard ratio="sin" name={"Sine"} tex={"\\Sin\\theta = y"} text={"<b>Vertical orange segment</b> from the <i>x</i>-axis to <i>P</i>. It's the point's height. Always between −1 and 1."} />
    <RatioCard ratio="cos" name={"Cosine"} tex={"\\Cos\\theta = x"} text={"<b>Horizontal purple segment</b> from the origin to <i>P</i>'s shadow on the <i>x</i>-axis. Always between −1 and 1."} />
    <RatioCard ratio="tan" name={"Tangent"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Yellow segment</b> on the line <i>x</i> = 1, which is tangent to the circle (hence the name). Undefined at 90° and 270°, because cos = 0."} />
    <RatioCard ratio="cot" name={"Cotangent"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Blue segment</b> on the line <i>y</i> = 1. Undefined at 0° and 180°, because sin = 0."} />
    <RatioCard ratio="sec" name={"Secant"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Red segment</b> from the origin to the end of the tangent. Its absolute value is never below 1."} />
    <RatioCard ratio="csc" name={"Cosecant"} tex={"\\Csc\\theta = 1 \\mathbin{/} \\Sin\\theta"} text={"<b>Green segment</b> from the origin to the end of the cotangent. Its absolute value is never below 1."} />
  </RatioGrid>
