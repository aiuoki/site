<Section eyebrow={"1 · Punto di partenza"} title={"Cos'è la circonferenza goniometrica"}>
  <Prose html={"È una circonferenza di <b>raggio 1</b> con centro nell'origine <span class=\"mr\">(0, 0)</span>. Un angolo <span class=\"m\">θ</span> si misura a partire dal semiasse <span class=\"m\">x</span> positivo, in senso antiorario. Dove il lato finale dell'angolo taglia la circonferenza c'è il punto <span class=\"m\">P</span>, e le sue coordinate sono proprio <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sen <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Grado"} text={"Un giro completo sono 360°."} />
    <DefItem title={"Radiante"} text={"Angolo il cui arco è lungo quanto il raggio. Un giro sono <span class=\"mr\">2π</span> rad. Per convertire: <span class=\"mr\">rad = gradi · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Quadrante"} text={"Ciascuna delle 4 zone in cui gli assi dividono il piano, numerate da I a IV in senso antiorario."} />
    <DefItem title={"Angolo di riferimento"} text={"L'angolo acuto tra il lato finale e l'asse <span class=\"m\">x</span>. Le sue funzioni sono quelle di <span class=\"m\">θ</span>, a meno del segno."} />
  </Defs>
</Section>

<Section eyebrow={"2 · Le sei funzioni"} title={"Cosa misura ogni segmento"} lead={"Poiché il raggio è 1, l'ipotenusa del triangolo vale 1 e ogni funzione diventa una lunghezza visibile."}>
  <RatioGrid>
    <RatioCard ratio="sin" name={"Seno"} tex={"\\Sin\\theta = y"} text={"<b>Segmento arancione verticale</b> dall'asse <i>x</i> fino a <i>P</i>. È l'altezza del punto. Sempre tra −1 e 1."} />
    <RatioCard ratio="cos" name={"Coseno"} tex={"\\Cos\\theta = x"} text={"<b>Segmento viola orizzontale</b> dall'origine fino all'ombra di <i>P</i> sull'asse <i>x</i>. Sempre tra −1 e 1."} />
    <RatioCard ratio="tan" name={"Tangente"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento giallo</b> sulla retta <i>x</i> = 1, tangente alla circonferenza (da qui il nome). Non esiste a 90° e 270°, perché cos = 0."} />
    <RatioCard ratio="cot" name={"Cotangente"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento blu</b> sulla retta <i>y</i> = 1. Non esiste a 0° e 180°, perché sen = 0."} />
    <RatioCard ratio="sec" name={"Secante"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento rosso</b> dall'origine fino all'estremo della tangente. Il suo valore assoluto non scende mai sotto 1."} />
    <RatioCard ratio="csc" name={"Cosecante"} tex={"\\Csc\\theta = 1 \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento verde</b> dall'origine fino all'estremo della cotangente. Il suo valore assoluto non scende mai sotto 1."} />
  </RatioGrid>
</Section>
