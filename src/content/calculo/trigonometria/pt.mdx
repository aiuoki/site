<Section eyebrow={"1 · Ponto de partida"} title={"O que é o círculo trigonométrico"}>
  <Prose html={"É uma circunferência de <b>raio 1</b> centrada na origem <span class=\"mr\">(0, 0)</span>. Um ângulo <span class=\"m\">θ</span> mede-se a partir do semieixo <span class=\"m\">x</span> positivo, rodando no sentido anti-horário. Onde o lado final do ângulo corta o círculo está o ponto <span class=\"m\">P</span>, e as suas coordenadas são exatamente <span class=\"mr\"><i>P</i> = (cos <i>θ</i>, sen <i>θ</i>)</span>."} />
  <Defs>
    <DefItem title={"Grau"} text={"Uma volta completa são 360°."} />
    <DefItem title={"Radiano"} text={"Ângulo cujo arco mede o mesmo que o raio. Uma volta são <span class=\"mr\">2π</span> rad. Para converter: <span class=\"mr\">rad = graus · π / 180</span>."} demo="rad" more="where" />
    <DefItem title={"Quadrante"} text={"Cada uma das 4 zonas em que os eixos dividem o plano, numeradas de I a IV no sentido anti-horário."} />
    <DefItem title={"Ângulo de referência"} text={"O ângulo agudo entre o lado final e o eixo <span class=\"m\">x</span>. As suas razões são as mesmas de <span class=\"m\">θ</span>, exceto o sinal."} />
  </Defs>
</Section>

<Section eyebrow={"2 · As seis razões"} title={"O que mede cada segmento"} lead={"Como o raio é 1, a hipotenusa do triângulo vale 1 e cada razão transforma-se num comprimento que se pode ver."}>
  <RatioGrid>
    <RatioCard ratio="sin" name={"Seno"} tex={"\\Sin\\theta = y"} text={"<b>Segmento laranja vertical</b> do eixo <i>x</i> até <i>P</i>. É a altura do ponto. Sempre entre −1 e 1."} />
    <RatioCard ratio="cos" name={"Cosseno"} tex={"\\Cos\\theta = x"} text={"<b>Segmento roxo horizontal</b> da origem até à sombra de <i>P</i> no eixo <i>x</i>. Sempre entre −1 e 1."} />
    <RatioCard ratio="tan" name={"Tangente"} tex={"\\Tan\\theta = \\Sin\\theta \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento amarelo</b> sobre a reta <i>x</i> = 1, que é tangente ao círculo (daí o nome). Não existe em 90° e 270°, porque cos = 0."} />
    <RatioCard ratio="cot" name={"Cotangente"} tex={"\\Cot\\theta = \\Cos\\theta \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento azul</b> sobre a reta <i>y</i> = 1. Não existe em 0° e 180°, porque sen = 0."} />
    <RatioCard ratio="sec" name={"Secante"} tex={"\\Sec\\theta = 1 \\mathbin{/} \\Cos\\theta"} text={"<b>Segmento vermelho</b> da origem até à extremidade da tangente. O seu valor absoluto nunca é menor que 1."} />
    <RatioCard ratio="csc" name={"Cossecante"} tex={"\\Csc\\theta = 1 \\mathbin{/} \\Sin\\theta"} text={"<b>Segmento verde</b> da origem até à extremidade da cotangente. O seu valor absoluto nunca é menor que 1."} />
  </RatioGrid>
</Section>

<Section eyebrow={"3 · Relações"} title={"Identidades fundamentais"} lead={"Uma identidade é uma igualdade verdadeira para qualquer ângulo (desde que as razões existam). Por baixo de cada uma vês a verificação com o ângulo atual; toca para ver a demonstração."}>
