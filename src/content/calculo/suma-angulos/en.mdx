<Section eyebrow={"2 · The formulas"} title={"Sum and difference"} lead={"Read <span class=\"mr\">±</span> with the top sign for a sum and the bottom one for a difference; <span class=\"mr\">∓</span> is the other way round. In the cosine the sign flips."}>
  <IdGroup>
    <IdCard demo="sum-sin" title={"Sine"} tex={"\\Sin(\\alpha \\pm \\beta) = \\Sin\\alpha\\,\\Cos\\beta \\pm \\Cos\\alpha\\,\\Sin\\beta"} text={"Same sign as the operation. Sine and cosine are mixed."} />
    <IdCard demo="sum-cos" title={"Cosine"} tex={"\\Cos(\\alpha \\pm \\beta) = \\Cos\\alpha\\,\\Cos\\beta \\mp \\Sin\\alpha\\,\\Sin\\beta"} text={"Opposite sign. cos goes with cos and sin with sin."} />
    <IdCard demo="sum-tan" title={"Tangent"} tex={"\\Tan(\\alpha \\pm \\beta) = (\\Tan\\alpha \\pm \\Tan\\beta) \\mathbin{/} (1 \\mp \\Tan\\alpha\\,\\Tan\\beta)"} text={"Comes from dividing the two above. Undefined if the denominator is 0."} />
  </IdGroup>
</Section>

<Section eyebrow={"3 · Worked example"} title={"sin 75° without a calculator"} lead={"75° isn't a key angle, but it is the sum of two: 45° + 30°."}>
  <Steps items={[{"text":"Split into key angles","tex":"75\\degree = 45\\degree + 30\\degree"},{"text":"Apply the sine formula","tex":"\\Sin 75\\degree = \\Sin 45\\degree \\cdot \\Cos 30\\degree + \\Cos 45\\degree \\cdot \\Sin 30\\degree"},{"text":"Substitute the table values","tex":"= \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{\\sqrt{3}}{2}\\right) + \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{1}{2}\\right)"},{"text":"Multiply and combine","tex":"= \\tfrac{\\sqrt{6}}{4} + \\tfrac{\\sqrt{2}}{4} = \\tfrac{\\sqrt{6} + \\sqrt{2}}{4} \\approx 0.9659"}]} />
</Section>
