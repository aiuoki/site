<Section eyebrow={"1 · A ideia"} title={"O que é somar ângulos"}>
  <Prose html={"Somar <span class=\"m\">α</span> (alfa) e <span class=\"m\">β</span> (beta) é rodar primeiro <span class=\"m\">α</span> a partir do eixo <span class=\"m\">x</span> e, daí, rodar mais <span class=\"m\">β</span>. Subtrair é rodar <span class=\"m\">β</span> para trás. As fórmulas da soma dão sen, cos e tg do ângulo final usando apenas os valores de <span class=\"m\">α</span> e <span class=\"m\">β</span>."} />
  <Warn title={"Erro típico: o seno não se distribui"} text={"sen(<i>α</i> + <i>β</i>) quase nunca é sen <i>α</i> + sen <i>β</i>. Com os ângulos atuais:"} live="wrong" />
</Section>

<Section eyebrow={"2 · As fórmulas"} title={"Soma e diferença"} lead={"Lê <span class=\"mr\">±</span> com o sinal de cima para a soma e o de baixo para a diferença; <span class=\"mr\">∓</span> é ao contrário. No cosseno o sinal inverte-se."}>
  <IdGroup>
    <IdCard demo="sum-sin" title={"Seno"} tex={"\\Sin(\\alpha \\pm \\beta) = \\Sin\\alpha\\,\\Cos\\beta \\pm \\Cos\\alpha\\,\\Sin\\beta"} text={"Mesmo sinal da operação. Misturam-se seno e cosseno."} />
    <IdCard demo="sum-cos" title={"Cosseno"} tex={"\\Cos(\\alpha \\pm \\beta) = \\Cos\\alpha\\,\\Cos\\beta \\mp \\Sin\\alpha\\,\\Sin\\beta"} text={"Sinal contrário. Junta-se cos com cos e sen com sen."} />
    <IdCard demo="sum-tan" title={"Tangente"} tex={"\\Tan(\\alpha \\pm \\beta) = (\\Tan\\alpha \\pm \\Tan\\beta) \\mathbin{/} (1 \\mp \\Tan\\alpha\\,\\Tan\\beta)"} text={"Obtém-se dividindo as duas anteriores. Não existe se o denominador for 0."} />
  </IdGroup>
</Section>

<Section eyebrow={"3 · Exemplo resolvido"} title={"sen 75° sem calculadora"} lead={"75° não é um ângulo notável, mas é a soma de dois: 45° + 30°."}>
  <Steps items={[{"text":"Decompor em ângulos notáveis","tex":"75\\degree = 45\\degree + 30\\degree"},{"text":"Aplicar a fórmula do seno","tex":"\\Sin 75\\degree = \\Sin 45\\degree \\cdot \\Cos 30\\degree + \\Cos 45\\degree \\cdot \\Sin 30\\degree"},{"text":"Substituir os valores da tabela","tex":"= \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{\\sqrt{3}}{2}\\right) + \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{1}{2}\\right)"},{"text":"Multiplicar e juntar","tex":"= \\tfrac{\\sqrt{6}}{4} + \\tfrac{\\sqrt{2}}{4} = \\tfrac{\\sqrt{6} + \\sqrt{2}}{4} \\approx 0{,}9659"}]} />
</Section>
