<Section eyebrow={"1 · La idea"} title={"Qué es sumar ángulos"}>
  <Prose html={"Sumar <span class=\"m\">α</span> (alfa) y <span class=\"m\">β</span> (beta) es girar primero <span class=\"m\">α</span> desde el eje <span class=\"m\">x</span> y, desde ahí, girar <span class=\"m\">β</span> más. Restar es girar <span class=\"m\">β</span> hacia atrás. Las fórmulas de suma calculan sen, cos y tan del ángulo final usando solo los valores de <span class=\"m\">α</span> y <span class=\"m\">β</span>."} />
  <Warn title={"Error típico: el seno no se reparte"} text={"sen(<i>α</i> + <i>β</i>) casi nunca es sen <i>α</i> + sen <i>β</i>. Con los ángulos actuales:"} live="wrong" />
</Section>

<Section eyebrow={"2 · Las fórmulas"} title={"Suma y resta"} lead={"Lee <span class=\"mr\">±</span> con el signo de arriba para la suma y el de abajo para la resta; <span class=\"mr\">∓</span> va al revés. En el coseno el signo se invierte."}>
  <IdGroup>
    <IdCard demo="sum-sin" title={"Seno"} tex={"\\Sin(\\alpha \\pm \\beta) = \\Sin\\alpha\\,\\Cos\\beta \\pm \\Cos\\alpha\\,\\Sin\\beta"} text={"Mismo signo que la operación. Se mezclan seno y coseno."} />
    <IdCard demo="sum-cos" title={"Coseno"} tex={"\\Cos(\\alpha \\pm \\beta) = \\Cos\\alpha\\,\\Cos\\beta \\mp \\Sin\\alpha\\,\\Sin\\beta"} text={"Signo contrario. Se agrupan cos con cos y sen con sen."} />
    <IdCard demo="sum-tan" title={"Tangente"} tex={"\\Tan(\\alpha \\pm \\beta) = (\\Tan\\alpha \\pm \\Tan\\beta) \\mathbin{/} (1 \\mp \\Tan\\alpha\\,\\Tan\\beta)"} text={"Sale de dividir las dos anteriores. No existe si el denominador es 0."} />
  </IdGroup>
</Section>

<Section eyebrow={"3 · Ejemplo resuelto"} title={"sen 75° sin calculadora"} lead={"75° no es un ángulo notable, pero sí es la suma de dos: 45° + 30°."}>
  <Steps items={[{"text":"Descomponer en notables","tex":"75\\degree = 45\\degree + 30\\degree"},{"text":"Aplicar la fórmula del seno","tex":"\\Sin 75\\degree = \\Sin 45\\degree \\cdot \\Cos 30\\degree + \\Cos 45\\degree \\cdot \\Sin 30\\degree"},{"text":"Sustituir los valores de la tabla","tex":"= \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{\\sqrt{3}}{2}\\right) + \\left(\\tfrac{\\sqrt{2}}{2}\\right)\\left(\\tfrac{1}{2}\\right)"},{"text":"Multiplicar y juntar","tex":"= \\tfrac{\\sqrt{6}}{4} + \\tfrac{\\sqrt{2}}{4} = \\tfrac{\\sqrt{6} + \\sqrt{2}}{4} \\approx 0{,}9659"}]} />
</Section>
