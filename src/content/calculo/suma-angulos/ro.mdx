  <Prose html={"A aduna <span class=\"m\">α</span> (alfa) și <span class=\"m\">β</span> (beta) înseamnă să rotești întâi cu <span class=\"m\">α</span> de la axa <span class=\"m\">x</span> și apoi încă cu <span class=\"m\">β</span>. A scădea înseamnă să rotești cu <span class=\"m\">β</span> înapoi. Formulele de adunare dau sin, cos și tg ale unghiului final folosind doar valorile lui <span class=\"m\">α</span> și <span class=\"m\">β</span>."} />

<Section eyebrow={"2 · Formulele"} title={"Suma și diferența"} lead={"Citește <span class=\"mr\">±</span> cu semnul de sus pentru sumă și cu cel de jos pentru diferență; la <span class=\"mr\">∓</span> e invers. La cosinus semnul se inversează."}>
  <IdGroup>
    <IdCard demo="sum-sin" title={"Sinus"} tex={"\\Sin(\\alpha \\pm \\beta) = \\Sin\\alpha\\,\\Cos\\beta \\pm \\Cos\\alpha\\,\\Sin\\beta"} text={"Același semn ca operația. Se amestecă sinusul și cosinusul."} />
    <IdCard demo="sum-cos" title={"Cosinus"} tex={"\\Cos(\\alpha \\pm \\beta) = \\Cos\\alpha\\,\\Cos\\beta \\mp \\Sin\\alpha\\,\\Sin\\beta"} text={"Semn opus. cos cu cos și sin cu sin."} />
    <IdCard demo="sum-tan" title={"Tangentă"} tex={"\\Tan(\\alpha \\pm \\beta) = (\\Tan\\alpha \\pm \\Tan\\beta) \\mathbin{/} (1 \\mp \\Tan\\alpha\\,\\Tan\\beta)"} text={"Se obține împărțind cele două de mai sus. Nu există dacă numitorul este 0."} />
  </IdGroup>
</Section>

<Section eyebrow={"3 · Exemplu rezolvat"} title={"sin 75° fără calculator"} lead={"75° nu este un unghi remarcabil, dar este suma a două: 45° + 30°."}>
